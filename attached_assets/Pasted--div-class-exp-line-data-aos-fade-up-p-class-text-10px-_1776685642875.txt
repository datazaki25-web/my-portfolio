<script>
    AOS.init({ duration: 1000, once: true });

    function setLang(lang) {
        document.getElementById('btn-id').classList.toggle('active', lang === 'id');
        document.getElementById('btn-en').classList.toggle('active', lang === 'en');
        const idElements = document.querySelectorAll('.lang-id-content');
        const enElements = document.querySelectorAll('.lang-en-content');
        if(lang === 'id') {
            idElements.forEach(el => el.style.display = 'block');
            enElements.forEach(el => el.style.display = 'none');
        } else {
            idElements.forEach(el => el.style.display = 'none');
            enElements.forEach(el => el.style.display = 'block');
        }
    }

</script>
